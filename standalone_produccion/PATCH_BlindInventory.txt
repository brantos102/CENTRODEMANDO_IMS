        if (DEEPLINK_FILE_ID && DEEPLINK_FILE_ID.length > 10) {
          // FIX 8.37: deep-link robusto + auto-login con el Gmail actual.
          CURRENT_FILE_ID = DEEPLINK_FILE_ID;
          const invEnc = ALL_INVENTORIES.find(inv => inv.id === DEEPLINK_FILE_ID);
          const invName = invEnc ? invEnc.link : (DEEPLINK_CLIENTE || "Inventario");
          if (DEEPLINK_CLIENTE) cargarBaseEAN(DEEPLINK_CLIENTE);
          try { google.script.run.registrarAccesoWMS(DEEPLINK_FILE_ID, DEEPLINK_CLIENTE, true); } catch(e) {}
          google.script.run.withSuccessHandler(function(s){
            if (s && s.autenticado && s.registrado && (s.rol === 'ADMIN' || s.rol === 'OPERADOR')) {
              USER_EMAIL = s.email; USER_ROLE = s.rol;
              iniciarSesionExitosa(s.nombre, invName);
            } else if (s && s.autenticado) {
              USER_EMAIL = s.email; USER_ROLE = 'GUEST';
              iniciarSesionExitosa(s.nombre + " (lectura)", invName);
            } else {
              USER_EMAIL = "INVITADO"; USER_ROLE = "GUEST";
              iniciarSesionExitosa("Auditor", invName);
            }
          }).withFailureHandler(function(){
            USER_EMAIL = "INVITADO"; USER_ROLE = "GUEST";
            iniciarSesionExitosa("Auditor", invName);
          }).obtenerSesionWMSActual();
          return;
        }
