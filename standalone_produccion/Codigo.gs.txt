        var nc = Math.max(5, sh.getLastColumn());

function obtenerBaseEAN(clienteActual) {
  try {
    let iter = DriveApp.searchFiles("title contains 'RL_PRODUCTO_CODIGOS EAN13'"); 
    if (!iter.hasNext()) return { error: "Archivo EAN no encontrado en Drive." };
    
    let file = null, lastUpdated = 0;
    while (iter.hasNext()) {
        let f = iter.next();
        if (f.getLastUpdated().getTime() > lastUpdated) { file = f; lastUpdated = f.getLastUpdated().getTime(); }
    }
    
    const csvContent = file.getBlob().getDataAsString();
    const delimiter = csvContent.indexOf(';') > -1 ? ';' : ',';
    const lines = csvContent.split(/\r?\n/);
    let eanMap = {};
    
    for (let i = 1; i < lines.length; i++) {
      let line = lines[i].trim();
      if (!line) continue;
      
      let parts = line.split(delimiter);
      if (parts.length >= 3) {
          let clienteDB = String(parts[0]).trim().toUpperCase(); 
          if (!clienteActual || clienteActual === 'TODOS' || clienteDB === String(clienteActual).trim().toUpperCase()) {
              let realSku = String(parts[1]).trim().toUpperCase(); 
              let codigoEAN = String(parts[2]).trim().toUpperCase(); 
              if (codigoEAN && realSku) eanMap[codigoEAN] = realSku; 
          }
      }
    }
    return { data: eanMap };
  } catch(e) { 
    return { error: "Error leyendo EAN: " + e.message }; 
  }
}

function registrarNuevoUsuario(data) {

    if (!responsable || responsable.pass !== data.passResponsable) return { exito: false, mensaje: "Contraseña del Responsable Incorrecta. Creación denegada." };
    if (usersDb[data.email]) return { exito: false, mensaje: "El correo ya está registrado." };
    
    let props = PropertiesService.getScriptProperties();
    let tempUsersStr = props.getProperty('WMS_TEMP_USERS');
    let tempUsers = tempUsersStr ? JSON.parse(tempUsersStr) : {};
    
    tempUsers[data.email] = { nombre: data.nombre, rol: "TEMPORAL", pass: data.pass, responsable: data.responsable, inventario_apoyo: data.inventario };

function onEdit(e) {
  try {
    if (!e || !e.range) return;
    const sheet = e.range.getSheet(), name = sheet.getName(), row = e.range.getRow(), col = e.range.getColumn();
    if (row < 2) return;

    if (name === "REGISTRO") {
       SpreadsheetApp.getUi().alert("⛔ ALERTA DE SEGURIDAD\n\nEl registro de auditoría es INALTERABLE.");
       if (e.oldValue !== undefined) { e.range.setValue(e.oldValue); } else { e.range.clearContent(); }
       return; 
    }
    if (name === "PLANILLA DE CONTEO FISICO") {
      if (row >= 3 && (col === 1 || col === 2 || col === 3)) {
        var nombreCol = col === 1 ? "FECHA INICIO" : col === 2 ? "FECHA FINAL" : "ID";
        SpreadsheetApp.getActiveSpreadsheet().toast(
          "⛔ " + nombreCol + " se genera automáticamente desde la fila 2. No editar aquí.",
          "Edición bloqueada", 3
        );
        if (e.oldValue !== undefined) { e.range.setValue(e.oldValue); }
        else { e.range.clearContent(); }
        return;
      }
      if (row === 2 && col === 2) {
        if (e.value !== "" && e.value !== null) {
          SpreadsheetApp.getActiveSpreadsheet().toast(
            "⛔ FECHA FINAL es automática (se calcula al cerrar el inventario).",
            "Bloqueado", 3
          );
          e.range.setValue(e.oldValue || "");
        }
        return;
      }

      if ([22, 23, 24].includes(col)) {
        if (e.oldValue !== undefined && e.oldValue !== "" && e.value !== e.oldValue) {
           SpreadsheetApp.getUi().alert(`⛔ CONTEO CERRADO\nSolicite un RECONTEO o AJUSTE.`);
           e.range.setValue(e.oldValue); return;
        }
        verificarYActualizarColumnaB(sheet);
        registrarAccionManual(e, sheet, row, col);
      }
      if (row === 2 && (col === 1 || col === 3)) { actualizarColumnasAC(sheet); generarSecuenciaColumnaD(sheet); }
      if (col === 7) consolidarDatos();
    }
  } catch (error) { }
}


function pollRealTime(posicionActual, emailUsuario, reqType, clientLastUpdate, fileId) {
  const cache = CacheService.getScriptCache();
  let activeUsers = [];
  const allUsers = getTodosLosUsuarios();

  if (emailUsuario && emailUsuario !== "INVITADO") {
    let usersObj = {};
    try { const cached = cache.get(`WMS_USERS_${fileId}`); if (cached) usersObj = JSON.parse(cached); } catch(e) {}
    usersObj[emailUsuario] = { t: Date.now(), pos: String(posicionActual || "") };
    const now = Date.now();
    for (let e in usersObj) {
      let info = usersObj[e]; if (typeof info === 'number') info = { t: info, pos: "" };
      if (now - info.t < 35000) activeUsers.push({ email: e, nombre: (allUsers[e] ? allUsers[e].nombre : e.split('@')[0]), pos: info.pos });
      else delete usersObj[e];
    }
    cache.put(`WMS_USERS_${fileId}`, JSON.stringify(usersObj), 60);
  }

  let serverLastUpdateStr = cache.get(`WMS_UPDATE_${fileId}`);
  let serverLastUpdate = serverLastUpdateStr ? parseInt(serverLastUpdateStr) : null;
  
  if (serverLastUpdate === null) {
      serverLastUpdate = Date.now();
      cache.put(`WMS_UPDATE_${fileId}`, serverLastUpdate.toString(), 21600);
  }
  
  let payload = { activeUsers: activeUsers, serverLastUpdate: serverLastUpdate };
  payload.depotVisible = PropertiesService.getScriptProperties().getProperty(`DEPOT_VIS_${fileId}`) === '1';

  if (clientLastUpdate !== 0 && serverLastUpdate <= clientLastUpdate) {
     payload.unchanged = true;
     return payload;
  }

  payload.unchanged = false;

  try {
      const ss = getTargetSS(fileId);
      const sheet = ss.getSheetByName("PLANILLA DE CONTEO FISICO");
      if (!sheet) throw new Error("Falta hoja PLANILLA DE CONTEO FISICO");
      
      const lastRow = sheet.getLastRow();
      if (lastRow < 2) {
          payload.avance = 0; payload.textoAvance = "0/0 Refs";
          if (reqType === 'HOME') payload.data = { exito: true, posiciones: [] };
          else if (reqType === 'COUNT') payload.data = { exito: true, items: [], posicion: posicionActual };
          else if (reqType === 'ANALYSIS') payload.data = { exito: true, kpis: { efectividadUnidades:0, efectividadRef:0, efectividadPos:0, uniValidadas:0, uniTotal:0, refValidadas:0, refTotal:0, posValidadas:0, posTotal:0 }, discrepancias: [] };
          return payload;
      }
      
      const fullData = sheet.getRange(2, 1, lastRow - 1, 29).getValues(); 

      let tRef = 0, cRef = 0;
      for(let i=0; i<fullData.length; i++) {
        const sku = String(fullData[i][6]).trim(); 
        const catLog = String(fullData[i][12]).trim().toUpperCase(); 
        const estadoFinal = String(fullData[i][19]).trim().toUpperCase(); 
        
        if (!sku || catLog === "DIF_INV") continue; 
        tRef++;
        if (fullData[i][21] !== "" || fullData[i][22] !== "" || fullData[i][23] !== "" || estadoFinal === "CORRECTO") cRef++;
      }
      payload.avance = tRef === 0 ? 0 : Math.round((cRef / tRef) * 100);
      payload.textoAvance = `${cRef} / ${tRef} Refs`;

      if (reqType === 'HOME') payload.data = parseHomeData(fullData);
      else if (reqType === 'COUNT' && posicionActual) payload.data = parseCountData(fullData, posicionActual);
      else if (reqType === 'ANALYSIS') payload.data = parseAnalysisData(fullData);

  } catch(e) { payload.data = { exito: false, error: e.message }; }

  return payload;
}

function parseHomeData(data) {
  const mapaPos = {};
  for(let i=0; i < data.length; i++) {
    const sku = String(data[i][6]).trim(); 
    const catLog = String(data[i][12]).trim().toUpperCase(); 
    const pos = String(data[i][14]).trim().toUpperCase(); 
    
    if (!pos || catLog === "DIF_INV" || !sku) continue; 

    const estadoFinal = String(data[i][19]).trim().toUpperCase(); 
    const c1 = data[i][21], c2 = data[i][22], c3 = data[i][23]; 
    
    if (!mapaPos[pos]) mapaPos[pos] = { nombre: pos, hasPendiente: false, hasCorrecto: false, hasFaltante: false, hasSobrante: false, totalItems: 0, validadosItems: 0 };

    mapaPos[pos].totalItems++;
    if (c1 !== "" || c2 !== "" || c3 !== "" || estadoFinal === "CORRECTO") mapaPos[pos].validadosItems++;

    if (estadoFinal === "PENDIENTE" || estadoFinal === "") mapaPos[pos].hasPendiente = true;
    else if (estadoFinal === "CORRECTO") mapaPos[pos].hasCorrecto = true;
    else if (estadoFinal === "FALTANTE") mapaPos[pos].hasFaltante = true;
    else if (estadoFinal === "SOBRANTE") mapaPos[pos].hasSobrante = true;
  }

  const posicionesArray = [];
  for (let p in mapaPos) {
     const m = mapaPos[p];
     let classColor = "", catName = "", priority = 99;
     let pctPos = m.totalItems === 0 ? 0 : Math.round((m.validadosItems / m.totalItems) * 100);
     
     if (m.hasFaltante && m.hasSobrante) { catName = "MIXTO (Falt/Sob)"; classColor = "var(--orange)"; priority = 1; }
     else if (m.hasFaltante) { catName = "FALTANTES"; classColor = "var(--danger)"; priority = 2; }
     else if (m.hasSobrante) { catName = "SOBRANTES"; classColor = "var(--warning)"; priority = 3; }
     else if (pctPos < 100 || m.hasPendiente) { catName = "EN PROCESO"; classColor = "var(--secondary)"; priority = 4; }
     else { catName = "COMPLETADAS"; classColor = "var(--success)"; priority = 5; }

     posicionesArray.push({ nombre: p, categoria: catName, color: classColor, prioridad: priority, avance: pctPos });
  }
  return { exito: true, posiciones: posicionesArray.sort((a,b) => a.prioridad - b.prioridad || a.nombre.localeCompare(b.nombre)) };
}

function parseCountData(data, target) {
  const items = [];
  for (let i = 0; i < data.length; i++) {
    if (String(data[i][14]).trim().toUpperCase() === target) {
      let step = 1;
      if(data[i][21] !== "") step=2; if(data[i][22] !== "") step=3; if(data[i][23] !== "") step=4;

      items.push({
        row: i + 2, codigo: data[i][6], desc: data[i][7], serie: data[i][8], lote: data[i][9],
        nDespacho: data[i][10], nPartida: data[i][11], 
        un: data[i][15], /* <--- ¡CORREGIDO! Índice 15 equivale exactamente a la Columna P */
        abc: data[i][5], 
        cat: data[i][12], est: data[i][13], depot: data[i][16],
        c1: data[i][21], c2: data[i][22], c3: data[i][23], nextStep: step, estado: data[i][19], 
        obs: data[i][27], newpos: data[i][28] 
      });
    }
  }
  return { exito: true, items: items, posicion: target };
}

function parseAnalysisData(data) {
  let qFisicaTotal = 0, qTeoricaTotal = 0, uCorrectas = 0, uContadasTeoricas = 0;
  let allRefs = new Set(), allPos = new Set(); 
  let refsCounted = new Set(), refsCorrect = new Set(), refsError = new Set();
  let posCounted = new Set(), posCorrect = new Set(), posError = new Set();
  let discrepancias = [];

  for(let r of data) {
    let sku = String(r[6]).trim(), desc = String(r[7]).trim(), pos = String(r[14]).trim(); 
    let qTeo = parseFloat(r[16]) || 0, qFis = parseFloat(r[17]) || 0, qDif = parseFloat(r[18]) || 0; 
    let estado = String(r[19]).trim().toUpperCase(), categoria = String(r[12]).trim().toUpperCase(); 

    if(!sku || categoria === "DIF_INV") continue;

    // EXTRACCIÓN DE DATOS DE TRAZABILIDAD (Columnas V a la AC de tu Excel)
    let conteo1 = String(r[21]).trim();
    let conteo2 = String(r[22]).trim();
    let conteo3 = String(r[23]).trim();
    let ajuste = String(r[24]).trim();
    let motivo = String(r[25]).trim();
    let justif = String(r[26]).trim();
    let obs = String(r[27]).trim();
    let nuevaPos = String(r[28]).trim();

    allRefs.add(sku); allPos.add(pos);
    qTeoricaTotal += qTeo; qFisicaTotal += qFis;

    if (estado !== "PENDIENTE" && estado !== "") {
       refsCounted.add(sku); posCounted.add(pos);
       uContadasTeoricas += qTeo; 
       
       // Empaquetamos toda la historia del artículo
       let itemData = { 
           sku: sku, desc: desc, pos: pos, qTeo: qTeo, qFis: qFis, dif: qDif, estado: estado,
           conteo1: conteo1, conteo2: conteo2, conteo3: conteo3, ajuste: ajuste, 
           motivo: motivo, justificacion: justif, obs: obs, nuevaPos: nuevaPos 
       };

       if (estado === "CORRECTO") { 
           uCorrectas += qFis; refsCorrect.add(sku); posCorrect.add(pos); 
           
           // REGLA NUEVA: Si está correcto pero fue por un ajuste o justificación, se va al reporte
           if (ajuste !== "" || motivo !== "" || justif !== "" || obs !== "") {
               itemData.estado = "CORR. AJUSTADO"; // Etiqueta especial para que no se confunda con un faltante real
               discrepancias.push(itemData);
           }
       } 
       else if (estado === "FALTANTE" || estado === "SOBRANTE") {
         refsError.add(sku); posError.add(pos);
         if (estado === "SOBRANTE") uCorrectas += qTeo;
         if (estado === "FALTANTE") uCorrectas += qFis;
         discrepancias.push(itemData);
       }
    }
  }

  refsError.forEach(sku => refsCorrect.delete(sku));
  posError.forEach(pos => posCorrect.delete(pos));

  return {
    exito: true,
    kpis: {
      efectividadUnidades: qTeoricaTotal === 0 ? 0 : ((uCorrectas / qTeoricaTotal) * 100).toFixed(1),
      efectividadRef: refsCounted.size === 0 ? 0 : ((refsCorrect.size / refsCounted.size) * 100).toFixed(1),
      efectividadPos: posCounted.size === 0 ? 0 : ((posCorrect.size / posCounted.size) * 100).toFixed(1),
      uniValidadas: uContadasTeoricas, uniTotal: qTeoricaTotal,
      refValidadas: refsCounted.size, refTotal: allRefs.size,
      posValidadas: posCounted.size, posTotal: allPos.size
    },
    discrepancias: discrepancias.sort((a,b) => a.pos.localeCompare(b.pos))
  };
}

      finder.forEach(cell => {

         const p = String(vals[0]).trim().toUpperCase();

            const dif = parseFloat(vals[4]) || 0;

function validarSerieCompleta(serie, fileId) {
  try {
    const ss = getTargetSS(fileId);
    const shSeries = ss.getSheetByName("PLANILLA DE SERIES FISICAS") || ss.getSheetByName("PLANILLA SERIES");
    const shFisico = ss.getSheetByName("PLANILLA DE CONTEO FISICO");
    const serieBuscada = String(serie).trim().toUpperCase();

    let duplicadoExacto = false;
    let posibleSubEquipo = null;  
    let filaDup = 0;

    if (shSeries && shSeries.getLastRow() > 1) {
      const dataSeries = shSeries.getRange(2, 2, shSeries.getLastRow() - 1, 1).getValues();
      for (let i = 0; i < dataSeries.length; i++) {
        const dbVal = String(dataSeries[i][0]).trim().toUpperCase();
        if (!dbVal) continue;
        if (dbVal === serieBuscada) {
          duplicadoExacto = true;
          filaDup = i + 2;
          break;
        }
        if (!posibleSubEquipo && _esSerieParecida(serieBuscada, dbVal)) {
          posibleSubEquipo = { fila: i + 2, serie: dbVal };
        }
      }
    }

    let datos = { sku: "SIN_DATOS", desc: "No registrado", pos: "---" };
    let encontrado = false;
    if (!duplicadoExacto && shFisico && shFisico.getLastRow() > 1) {
      const dataFis = shFisico.getRange(2, 7, shFisico.getLastRow() - 1, 9).getValues();
      for (let i = 0; i < dataFis.length; i++) {
        const dbSerie = String(dataFis[i][2]).trim().toUpperCase();
        if (dbSerie === serieBuscada) {       
          encontrado = true;
          datos = {
            sku:  String(dataFis[i][0]).trim(),
            desc: String(dataFis[i][1]).trim(),
            pos:  String(dataFis[i][8]).trim()
          };
          break;
        }
      }
    }

    return {
      duplicado: duplicadoExacto,
      fila: filaDup,
      encontrado: encontrado,
      datos: datos,
      serieReal: serieBuscada,
      posibleSubEquipo: posibleSubEquipo  
    };
  } catch (e) {
    return { duplicado: false, encontrado: false, datos: {}, serieReal: serie };
  }
}

function _esSerieParecida(a, b) {
  if (!a || !b || a === b) return false;
  if (a.length < 6 || b.length < 6) return false;
  const corta = a.length < b.length ? a : b;
  const larga = a.length < b.length ? b : a;
  if (larga.length - corta.length <= 2 && larga.startsWith(corta)) return true;
  if (a.length === b.length) {
    let diffs = 0;
    for (let i = 0; i < a.length; i++) {
      if (a[i] !== b[i]) { diffs++; if (diffs > 1) return false; }
    }
    return diffs === 1;
  }
  return false;
}

function procesarGuardadoInventario(datos, datosHallazgo, usuarioOperador, fileId) {

  try {

      const ss = getTargetSS(fileId);
      const shFisico = ss.getSheetByName("PLANILLA DE CONTEO FISICO");
      let shSeries = ss.getSheetByName("PLANILLA DE SERIES FISICAS") || ss.getSheetByName("PLANILLA SERIES");

      const shAlterna = ss.getSheetByName("PLANILLA ALTERNA");

      let maxSeq = 0;
      if (realLastRow > 1) {
        const ids = shSeries.getRange(2, 1, realLastRow - 1, 1).getValues();
        ids.forEach(r => { const s = Number(r[0]); if (!isNaN(s) && s > maxSeq) maxSeq = s; });
      }

      const seriesExistentes = new Set();
      const dataSeries = shSeries.getRange(2, 2, realLastRow || 1, 1).getValues();
      dataSeries.forEach(r => { if(r[0]) seriesExistentes.add(String(r[0]).trim().toUpperCase()); });

      const mapaSeriesRow = {}; 
      if (shFisico.getLastRow() > 1) {
        const dataFisico = shFisico.getRange(2, 3, shFisico.getLastRow()-1, 18).getValues(); 
        dataFisico.forEach((r, i) => { if(r[6]) mapaSeriesRow[String(r[6]).trim().toUpperCase()] = { fila: i + 2, id: r[0], cliente: r[2], sku: r[4], pos: r[12], estado: r[17] }; });
      }

      const registrosAB = [], registrosDEF = [];
      const registrosAlterna = [], logAuditoria = [], conteosSumar = {}, duplicados = [];
      const user = usuarioOperador || Session.getActiveUser().getEmail(); 
      const timestamp = new Date();

      datos.forEach(item => {
        const serie = String(item.serie).trim().toUpperCase(), pos = String(item.posicion).trim();
        let sku = item.producto; 

        if (seriesExistentes.has(serie)) {
          duplicados.push(serie);
        } else {
          maxSeq++; seriesExistentes.add(serie);
          let logId = "SERIE_ADD", logCliente = "APP_WMS", logSku = sku, logPos = pos, logEstado = "PENDIENTE";

          registrosAB.push([maxSeq, serie]);
          registrosDEF.push([sku, pos, user]); 
          
          if (mapaSeriesRow[serie]) {
            const info = mapaSeriesRow[serie];
            if (!conteosSumar[info.fila]) {
                conteosSumar[info.fila] = { cantidad: 0, nueva_pos: item.nueva_posicion || "" };
            }
            conteosSumar[info.fila].cantidad += 1;
            if (item.nueva_posicion) conteosSumar[info.fila].nueva_pos = item.nueva_posicion;

            logId = info.id; logCliente = info.cliente; logSku = info.sku; logPos = info.pos; logEstado = info.estado; sku = info.sku; 
          } else if (item.esDesconocido && datosHallazgo && shAlterna) {
            registrosAlterna.push([timestamp, user, "", datosHallazgo.sku, datosHallazgo.desc, serie, "", pos, 1, "", "", datosHallazgo.obs || "Hallazgo", datosHallazgo.justificacion]);
            logEstado = "HALLAZGO"; sku = datosHallazgo.sku;
          }
          
          let observacionAudi = serie;
          if(item.nueva_posicion) observacionAudi += " | Destino: " + item.nueva_posicion;

          logAuditoria.push([timestamp, "", "", "", user, logId, logCliente, logSku, logPos, 1, logEstado, 1, "", "", observacionAudi]);
        }
      });

      if (registrosAB.length > 0) {
         if (realLastRow + registrosAB.length > shSeries.getMaxRows()) shSeries.insertRowsAfter(shSeries.getMaxRows(), registrosAB.length + 20); 
         shSeries.getRange(realLastRow + 1, 1, registrosAB.length, 2).setValues(registrosAB);
         shSeries.getRange(realLastRow + 1, 4, registrosDEF.length, 3).setValues(registrosDEF);
      }
      
      if (registrosAlterna.length > 0 && shAlterna) shAlterna.getRange(shAlterna.getLastRow()+1, 1, registrosAlterna.length, 13).setValues(registrosAlterna);
      if (logAuditoria.length > 0 && shRegistro) { const r = shRegistro; r.getRange(r.getLastRow()+1, 1, logAuditoria.length, 15).setValues(logAuditoria); }

      if (Object.keys(conteosSumar).length > 0) {
        for (const [fila, dataSuma] of Object.entries(conteosSumar)) {
          const vals = shFisico.getRange(Number(fila), 22, 1, 3).getValues()[0];
          let targetCol = 22; 
          if(vals[0]==="" || vals[0]===null) targetCol=22; 
          else if(vals[1]==="" || vals[1]===null) targetCol=23; 
          else targetCol=24; 
          
          let curr = parseFloat(shFisico.getRange(Number(fila), targetCol).getValue()) || 0;
          shFisico.getRange(Number(fila), targetCol).setValue(curr + dataSuma.cantidad); 
          
          if(dataSuma.nueva_pos && dataSuma.nueva_pos !== "") {
              shFisico.getRange(Number(fila), 29).setValue(dataSuma.nueva_pos);
          }
        }
      }

      SpreadsheetApp.flush(); notificarCambioBD(fileId);
      return { exito: true, guardados: registrosAB.length, alternaGuardados: registrosAlterna.length, duplicados: duplicados };

  } catch(e) { return { exito: false, mensaje: "Error del Servidor: " + e.message }; 
  } finally { lock.releaseLock(); }
}

    var ss = SpreadsheetApp.openById(MASTER_DB_ID);
    var cron = ss.getSheetByName("CRONOGRAMA - 2026");
    if (!cron) { var all = ss.getSheets(); for (var a=0;a<all.length;a++){ if (all[a].getName().toUpperCase().indexOf("CRONOGRAMA")===0){ cron=all[a]; break; } } }
    if (!cron) return res;
    var lastR = cron.getLastRow();
    if (lastR < 2) return res;
    var disp = cron.getRange(2, 17, lastR - 1, 1).getDisplayValues(); // col Q, texto visible (rápido)
    for (var i = 0; i < disp.length; i++) {
      var txt = String(disp[i][0] || "").trim().toUpperCase();
      if (!txt || txt.length < 4) continue;
      var match = false;
      if (nA && (txt === nA || nA.indexOf(txt) !== -1 || txt.indexOf(nA) !== -1)) match = true;
      else if (fid && txt.indexOf(fid) !== -1) match = true;
      if (match) {
        var fila = i + 2;
        cron.getRange(fila, 13).setValue("Entregado"); // M = Estado
        cron.getRange(fila, 14).setValue(soloFecha);   // N = Fecha de entrega
        res.encontrado = true; res.fila = fila;
        break;
      }
    }
  } catch (e) { res.error = e.message; }
  return res;
}

function finalizarInventarioWMS(fileId, confirmar, fechaEntrega) {
  try {
    var ss = getTargetSS(fileId);
    var sheet = ss.getSheetByName("PLANILLA DE CONTEO FISICO");
    if (!sheet) return { exito:false, mensaje:"Falta PLANILLA DE CONTEO FISICO" };
    var lastRow = sheet.getLastRow(), tRef=0, cRef=0;
    if (lastRow >= 2) {
      var d = sheet.getRange(2,1,lastRow-1,24).getValues();
      for (var i=0;i<d.length;i++){
        var sku=String(d[i][6]).trim(), cat=String(d[i][12]).trim().toUpperCase(), est=String(d[i][19]).trim().toUpperCase();
        if(!sku||cat==="DIF_INV")continue; tRef++;
        if(d[i][21]!==""||d[i][22]!==""||d[i][23]!==""||est==="CORRECTO")cRef++;
      }
    }
    var avance = tRef===0?0:Math.round((cRef/tRef)*100);
    if (avance < 100 && !confirmar) return { exito:false, requiereConfirmacion:true, avance:avance, texto:cRef+" / "+tRef+" Refs" };
    // Fecha de entrega: editable. Si llega "yyyy-MM-dd" válida se usa; si no, hoy.
    var soloFecha = null;
    if (fechaEntrega) { var p = String(fechaEntrega).split("-"); if (p.length===3){ var dd=new Date(parseInt(p[0],10), parseInt(p[1],10)-1, parseInt(p[2],10)); if(!isNaN(dd.getTime())) soloFecha=dd; } }
    if (!soloFecha) { var hoy=new Date(); soloFecha=new Date(hoy.getFullYear(),hoy.getMonth(),hoy.getDate()); }
    sheet.getRange(2,2).setValue(soloFecha);  // sin setNumberFormat: la col puede tener validacion
    var panelOk=false, nombreArchivo="";
    try {
      var pan = SpreadsheetApp.openById(MASTER_DB_ID).getSheetByName("PANEL DE CONTROL");
      if (pan && pan.getLastRow()>1) {
        var pr = pan.getRange(2,1,pan.getLastRow()-1,3).getDisplayValues(); // A=cliente,B=nombre,C=fileId
        for (var k=0;k<pr.length;k++){ if(String(pr[k][2]).trim()===String(fileId).trim()){ nombreArchivo=String(pr[k][1]).trim(); pan.getRange(k+2,7).setValue("Entregado"); panelOk=true; break; } }
      }
    } catch(e){}
    if (!nombreArchivo) { try { nombreArchivo = DriveApp.getFileById(String(fileId).trim()).getName(); } catch(e){} }
    var cronRes = _actualizarCronogramaEntrega(fileId, soloFecha, nombreArchivo);
    notificarCambioBD(fileId);
    return { exito:true, avance:avance, texto:cRef+" / "+tRef+" Refs",
             fechaFin:Utilities.formatDate(soloFecha, Session.getScriptTimeZone(), "dd/MM/yyyy"),
             panelActualizado:panelOk, cronograma:cronRes.encontrado, cronArchivo:cronRes.archivo||nombreArchivo||"", cronError:cronRes.error||"" };
  } catch(e){ return { exito:false, mensaje:e.message }; }
}
