    function playSound(type) {
        if(isMuted) return;
        if (audioCtx.state === 'suspended') audioCtx.resume();
        const oscillator = audioCtx.createOscillator();
        const gainNode = audioCtx.createGain();
        oscillator.connect(gainNode);
        gainNode.connect(audioCtx.destination);

        if (type === 'beep') { 
            oscillator.type = 'sine'; oscillator.frequency.setValueAtTime(880, audioCtx.currentTime); 
            gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
            oscillator.start(); gainNode.gain.exponentialRampToValueAtTime(0.00001, audioCtx.currentTime + 0.1);
            oscillator.stop(audioCtx.currentTime + 0.1);
        } else if (type === 'success') { 
            oscillator.type = 'sine'; oscillator.frequency.setValueAtTime(523.25, audioCtx.currentTime); oscillator.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.1); 
            gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
            oscillator.start(); gainNode.gain.exponentialRampToValueAtTime(0.00001, audioCtx.currentTime + 0.3);
            oscillator.stop(audioCtx.currentTime + 0.3);
        } else if (type === 'error') { 
            oscillator.type = 'sawtooth'; oscillator.frequency.setValueAtTime(150, audioCtx.currentTime);
            gainNode.gain.setValueAtTime(0.2, audioCtx.currentTime);
            oscillator.start(); gainNode.gain.exponentialRampToValueAtTime(0.00001, audioCtx.currentTime + 0.3);
            oscillator.stop(audioCtx.currentTime + 0.3);
        }
    }

    function iniciarPolling() {
        if (stopPolling || IS_USER_TYPING || !CURRENT_FILE_ID) {
            if(!stopPolling) pollTimer = setTimeout(iniciarPolling, 4000);
            return;
        }

        let reqType = '';
        if (document.getElementById('page-home').classList.contains('active')) reqType = 'HOME';
        else if (document.getElementById('page-count').classList.contains('active') && CURRENT_POS) reqType = 'COUNT';
        else if (document.getElementById('page-analysis').classList.contains('active')) reqType = 'ANALYSIS';

        google.script.run.withSuccessHandler(res => {
            if (stopPolling) return;
            if (res.activeUsers) {
                ACTIVE_USERS_DATA = res.activeUsers;
                document.getElementById('activeUsersCount').innerText = res.activeUsers.length;
                document.getElementById('activeUsersCount').title = res.activeUsers.map(function(u){ return (typeof u === 'string') ? u : (u.nombre || u.email || ''); }).join("\n");

                  if (_sig !== LAST_AVATARS_SIG) { LAST_AVATARS_SIG = _sig; if (document.getElementById('page-home').classList.contains('active')) actualizarAvatares(); }
                } catch(e){}
            }
            if (!res.unchanged && res.data && res.data.exito !== false) {
                LOCAL_LAST_UPDATE = res.serverLastUpdate;
                
                if (res.depotVisible !== undefined) {
                    GLOBAL_DEPOT_VISIBLE = res.depotVisible;
                    applyDepotVisibility();
                }

                if(res.avance !== undefined) actualizarBarraProgreso(res.avance, res.textoAvance);

                if (reqType === 'HOME') { ALL_POS = res.data.posiciones || []; filterPositions(); } 
                else if (reqType === 'COUNT') { TABLE_DATA = res.data.items || []; renderTableData(); } 
                else if (reqType === 'ANALYSIS') { renderAnalysis(res.data); }
            }
            pollTimer = setTimeout(iniciarPolling, 4000);
        }).withFailureHandler(err => { 
            if(!stopPolling) pollTimer = setTimeout(iniciarPolling, 8000); 
        }).pollRealTime(CURRENT_POS, USER_EMAIL, reqType, LOCAL_LAST_UPDATE, CURRENT_FILE_ID);
    }

    function forceSync() { 
        LOCAL_LAST_UPDATE = 0; stopPolling = false; 
        clearTimeout(pollTimer); 
        
        let reqType = '';
        if (document.getElementById('page-home').classList.contains('active')) reqType = 'HOME';
        else if (document.getElementById('page-count').classList.contains('active') && CURRENT_POS) reqType = 'COUNT';
        else if (document.getElementById('page-analysis').classList.contains('active')) reqType = 'ANALYSIS';

        google.script.run.withSuccessHandler(res => {
            if (res && res.data && res.data.exito !== false) {
                if(res.avance !== undefined) actualizarBarraProgreso(res.avance, res.textoAvance);
                if (reqType === 'HOME') { ALL_POS = res.data.posiciones || []; filterPositions(); } 
                else if (reqType === 'COUNT') { TABLE_DATA = res.data.items || []; renderTableData(); } 
                else if (reqType === 'ANALYSIS') { renderAnalysis(res.data); }
            } else if (res && res.data && res.data.exito === false) {
                Swal.fire("Error de Hoja", res.data.error || "No se pudo leer la hoja.", "error");
            }
            iniciarPolling();
        }).withFailureHandler(err => {
            setTimeout(forceSync, 3000); 
        }).pollRealTime(CURRENT_POS, USER_EMAIL, reqType, 0, CURRENT_FILE_ID);
    }

    function cargarBaseEAN(clienteStr) {
       google.script.run.withSuccessHandler(res => { 
           if(res && res.data && Object.keys(res.data).length > 0) { 
               EAN_DATABASE = res.data; 
               showToast("Base de códigos EAN cargada exitosamente."); 
               
               // NUEVO: Llenar el autocompletado del buscador
               const datalist = document.getElementById('skuSuggestions');
               if (datalist) {
                   datalist.innerHTML = ""; // Limpiar lista anterior
                   const uniqueCodes = new Set();
                   
                   // Recopilamos todos los EAN y SKUs sin repetir
                   for (let ean in EAN_DATABASE) {
                       if (ean) uniqueCodes.add(ean);
                       if (EAN_DATABASE[ean]) uniqueCodes.add(EAN_DATABASE[ean]);
                   }
                   
                   // Inyectamos las opciones de golpe para que no se trabe la pantalla
                   let fragment = document.createDocumentFragment();
                   uniqueCodes.forEach(code => {
                       let option = document.createElement('option');
                       option.value = code;
                       fragment.appendChild(option);
                   });
                   datalist.appendChild(fragment);
               }
           }
       }).obtenerBaseEAN(clienteStr);
    }

    function entregarInventarioWMS() {
        if (USER_ROLE === 'GUEST') { return checkAuth(); }
        if (!CURRENT_FILE_ID || CURRENT_FILE_ID === 'LOCAL') { return Swal.fire("Sin inventario","Primero selecciona un inventario.","warning"); }
        const ejecutar = (confirmar) => {
            Swal.fire({ title: confirmar ? 'Confirmando entrega...' : 'Verificando avance...', didOpen: () => Swal.showLoading() });
            google.script.run.withSuccessHandler(res => {
                if (res && res.requiereConfirmacion) {
                    Swal.fire({ title:'Avance incompleto ('+res.avance+'%)',
                        html:'El inventario esta en <b>'+res.texto+'</b> ('+res.avance+'%).<br>Confirmas igualmente la <b>ENTREGA</b>?',
                        icon:'warning', showCancelButton:true, confirmButtonText:'Si, entregar', cancelButtonText:'Cancelar', confirmButtonColor:'#16a34a'
                    }).then(r => { if (r.isConfirmed) ejecutar(true); });
                    return;
                }
                if (res && res.exito) {
                    forceSync();
                    Swal.fire({ title:'Inventario Entregado',
                        html:'Fecha fin: <b>'+res.fechaFin+'</b><br>Avance: <b>'+res.avance+'%</b>'+(res.panelActualizado?'<br><span style="color:#16a34a">Panel actualizado</span>':''),
                        icon:'success', showCancelButton:true, confirmButtonText:'Generar correo de finalizacion', cancelButtonText:'Cerrar', confirmButtonColor:'#2563eb'
                    }).then(r => { if (r.isConfirmed) generarCorreoFinalWMS(); });
                } else { Swal.fire("Error", (res && res.mensaje) || "No se pudo registrar la entrega.", "error"); }
            }).withFailureHandler(e => Swal.fire("Error de Red", e.message, "error")).finalizarInventarioWMS(CURRENT_FILE_ID, confirmar);
        };
        ejecutar(false);
    }

    function generarCorreoFinalWMS() {

    }

    function handlePhotoUpload(rowId, input) {
        if(input.files && input.files[0]) {
            const reader = new FileReader();
            reader.onload = function(e) {
                const b64 = e.target.result;
                if(!UNSAVED_ROWS[rowId]) {
                    UNSAVED_ROWS[rowId] = { qty: document.getElementById('qty-'+rowId)?.value||"", obs: document.getElementById('obs-'+rowId)?.value||"", newPos: document.getElementById('newPos-'+rowId)?.value||"" };
                    document.getElementById('row-'+rowId).style.borderLeft = "6px solid var(--orange)";
                }
                UNSAVED_ROWS[rowId].photoB64 = b64;
                
                const icon = document.getElementById('cam-icon-' + rowId);
                icon.style.color = 'var(--accent)';
                
                showToast("Evidencia adjuntada localmente");
            };
            reader.readAsDataURL(input.files[0]);
        }
    }

    function renderTableData() {
      document.getElementById('loadingCount').style.display = 'none';
      const hideCorrect = document.getElementById('toggleCorrect').checked; 
      const filterText = (document.getElementById('tableFilter')?.value || "").toLowerCase();
      
      let displayData = TABLE_DATA.filter(item => {
          if (hideCorrect && item.estado === "CORRECTO") return false;
          if (filterText) {
              const searchString = (item.codigo + " " + (item.desc || "") + " " + (item.cat || "")).toLowerCase();
              if (!searchString.includes(filterText)) return false;
          }
          return true;
      });

      displayData.sort((a, b) => {
         if (a.estado === "CORRECTO" && b.estado !== "CORRECTO") return 1; 
         if (a.estado !== "CORRECTO" && b.estado === "CORRECTO") return -1;
         let valA = String(a[CONFIG_COUNT.sortKey]||"").toLowerCase(), valB = String(b[CONFIG_COUNT.sortKey]||"").toLowerCase();
         if (valA < valB) return CONFIG_COUNT.sortAsc ? -1 : 1; if (valA > valB) return CONFIG_COUNT.sortAsc ? 1 : -1; return 0;
      });

      filteredTableData = displayData;
      currentPage = 1;
      renderTablePage();
    }

    function changePage(dir) {
        const totalPages = Math.ceil(filteredTableData.length / ROWS_PER_PAGE) || 1;
        
        if (dir === 1 && currentPage >= totalPages) { saltarPosicion(1); return; }
        if (dir === -1 && currentPage <= 1) { saltarPosicion(-1); return; }
        
        currentPage += dir;
        renderTablePage();
    }

    function saltarPosicion(dir) {
        if (!ALL_POS || ALL_POS.length === 0 || !CURRENT_POS) return;
        
        let sortedPos = ALL_POS.map(p => p.nombre).sort((a,b) => a.localeCompare(b));
        let currentIndex = sortedPos.indexOf(CURRENT_POS);
        
        if (currentIndex !== -1) {
            let newIndex = currentIndex + dir;
            if (newIndex >= 0 && newIndex < sortedPos.length) {
                let nextPos = sortedPos[newIndex];
                seleccionarPosicion(nextPos);
                nav('count'); 
                loadCountTable(); 
                showToast("Cambiando a ubicación " + nextPos);
            } else {
                showToast(dir > 0 ? "Última posición alcanzada" : "Primera posición alcanzada");
            }
        }
    }

    function parsePos(posStr) {
       let parts = posStr.split('-');
       let gaveta = null;
       let nave = "";
       let literal = "";
       let col = 0, nivel = 0;
       let isFloor = posStr.toUpperCase().includes('PISO') || posStr.toUpperCase().includes('BPB');

       if (parts.length >= 5 && !isNaN(parts[parts.length-1])) {
           gaveta = parseInt(parts.pop(), 10) || null;
       }
       
       if (parts.length >= 4) {
           nivel = parseInt(parts.pop(), 10) || 0;
           col = parseInt(parts.pop(), 10) || 0;
           literal = parts.pop() || "";
           nave = parts.join('-') || "";
       } else {
           nave = posStr;
       }
       
       let isMZ = nave.toUpperCase().startsWith('MZ');
       let isGavetaArea = gaveta !== null; 
       
       return { nave, literal, col, nivel, gaveta, isMZ, isFloor, isGavetaArea };
    }

    function filterPositionsData() {
        let n = document.getElementById('fltNave')?.value.toUpperCase() || "";
        let r = document.getElementById('fltRack')?.value.toUpperCase() || "";
        let cRaw = document.getElementById('fltCol')?.value.trim() || "";
        let cNum = cRaw !== "" ? parseInt(cRaw, 10) : null;
        let nvRaw = document.getElementById('fltNiv')?.value.trim() || "";
        let nvNum = nvRaw !== "" ? parseInt(nvRaw, 10) : null;
        
        return ALL_POS.filter(p => {
            let parts = parsePos(p.nombre);
            let matchC = (cNum === null) || (parts.col === cNum) || String(parts.col).includes(cRaw) || p.nombre.includes(`-${cRaw}-`) || p.nombre.includes(`-${String(cRaw).padStart(2, '0')}-`);
            let matchNv = (nvNum === null) || (parts.nivel === nvNum) || String(parts.nivel).includes(nvRaw) || p.nombre.endsWith(`-${nvRaw}`);
            return parts.nave.includes(n) && parts.literal.includes(r) && matchC && matchNv;
        });
    }

    function setCameraView(view) {
        if(!GLOBAL_CAMERA || !GLOBAL_CONTROLS) return;
        const w = GLOBAL_ASSEMBLY_WIDTH || 20;
        const y = (GLOBAL_MAX_Y || 10) / 2;
        
        const startXFocus = -w / 2 + 1.6; 

        switch(view) {
            case 'top': 
                GLOBAL_CAMERA.position.set(0, Math.max(30, w * 0.8), 0); 
                GLOBAL_CONTROLS.target.set(0, 0, 0);
                isFlying = false;
                break;
            case 'front': 
                GLOBAL_CAMERA.position.set(0, y, Math.max(35, w * 0.9)); 
                GLOBAL_CONTROLS.target.set(0, y, 0);
                isFlying = false;
                break;
            case 'side': 
                GLOBAL_CAMERA.position.set(startXFocus - 15, y, 0); 
                GLOBAL_CONTROLS.target.set(startXFocus, y, 0);
                isFlying = false;
                break;
            case 'reset': 
                GLOBAL_CAMERA.position.set(startXFocus, y + 2, 22); 
                GLOBAL_CONTROLS.target.set(startXFocus, y, 0);
                isFlying = false;
                break;
        }
        GLOBAL_CONTROLS.update();
    }

    function update3DMaterials() {
        if(!objetosClickeables) return;
        
        let isXray = (current3DMode === 'XRAY');
        
        if(GLOBAL_MAT_POSTE) { GLOBAL_MAT_POSTE.transparent = isXray; GLOBAL_MAT_POSTE.opacity = isXray ? 0.15 : 1; }
        if(GLOBAL_MAT_VIGA) { GLOBAL_MAT_VIGA.transparent = isXray; GLOBAL_MAT_VIGA.opacity = isXray ? 0.15 : 1; }
        if(GLOBAL_MAT_REPISA) { GLOBAL_MAT_REPISA.transparent = isXray; GLOBAL_MAT_REPISA.opacity = isXray ? 0.15 : 1; }
        if(GLOBAL_FLOOR) { GLOBAL_FLOOR.material.transparent = isXray; GLOBAL_FLOOR.material.opacity = isXray ? 0.05 : 1; }
        
        objetosClickeables.forEach(obj => {
            const mesh = obj.userData.esGaveta ? obj.children[1] : obj;
            const cat = obj.userData.categoria;
            
            if(current3DMode === 'PROGRESS') {
                mesh.material.color.set(colorHexMap[cat] || '#94a3b8');
                mesh.material.opacity = 1; mesh.material.transparent = false;
                mesh.material.emissiveIntensity = 0;
            } else if (current3DMode === 'ALERTS') {
                if(cat === 'FALTANTES' || cat === 'SOBRANTES' || cat === 'MIXTO (Falt/Sob)') {
                    mesh.material.color.set(colorHexMap[cat]);
                    mesh.material.opacity = 1; mesh.material.transparent = false;
                    mesh.material.emissiveIntensity = 0;
                } else {
                    mesh.material.color.set('#cbd5e1');
                    mesh.material.opacity = 0.15; mesh.material.transparent = true;
                    mesh.material.emissiveIntensity = 0;
                }
            } else if (current3DMode === 'OCCUPATION') {
                mesh.material.color.set('#38bdf8'); 
                mesh.material.opacity = 1; mesh.material.transparent = false;
                mesh.material.emissiveIntensity = 0;
            } else if (current3DMode === 'XRAY') {
                if(['FALTANTES', 'SOBRANTES', 'MIXTO (Falt/Sob)'].includes(cat)) {
                    mesh.material.color.set(colorHexMap[cat]);

                    mesh.material.opacity = 0.05; mesh.material.transparent = true;

    // MEJORA VISUAL: helper de rectángulo redondeado (canvas), compatible con todo navegador.
    function _roundRectPath(ctx, x, y, w, h, r) {
        if (r > w/2) r = w/2; if (r > h/2) r = h/2;
        ctx.beginPath();
        ctx.moveTo(x+r, y);
        ctx.arcTo(x+w, y, x+w, y+h, r);
        ctx.arcTo(x+w, y+h, x, y+h, r);
        ctx.arcTo(x, y+h, x, y, r);
        ctx.arcTo(x, y, x+w, y, r);
        ctx.closePath();
    }

    function triggerQuickScan(){

       const qVal = parseFloat(qInputStr) || 1;
       if(!rawScan) { isQuickScanning = false; return; }

       let realSku = rawScan;
       let strippedScan = rawScan.replace(/^0+/, ''); 

       if (EAN_DATABASE) {
           if (EAN_DATABASE[rawScan]) { realSku = EAN_DATABASE[rawScan]; } 
           else {
               for (let key in EAN_DATABASE) {
                   if (key.replace(/^0+/, '') === strippedScan) { realSku = EAN_DATABASE[key]; break; }
               }
           }
       }
       
       let strippedRealSku = realSku.replace(/^0+/, '');
       const matches = TABLE_DATA.filter(i=>String(i.codigo).trim().toUpperCase()===realSku || String(i.codigo).trim().toUpperCase().replace(/^0+/, '') === strippedRealSku);
       
       if(matches.length === 0){

          google.script.run.withSuccessHandler(r=>{
              if(r.length>0) Swal.fire("Ubicación Incorrecta",`SKU detectado: ${realSku}.<br>Revisar Búsqueda SKU`,"info");
              else Swal.fire("No encontrado",`SKU (${realSku}) inexistente en este archivo.`,"error");
              resetQuickScanUI();
          }).buscarUbicacionesPorSku(realSku, CURRENT_FILE_ID);
       } else if(matches.length > 1) {
          playSound('beep');
          let html = `<div style="font-size:11px; text-align:left;"><table style="width:100%; border-collapse:collapse;"><tr><th>Sel</th><th>Lote</th><th>Partida</th><th>Cant. Dep</th></tr>`;

    function ejecutarSuma(item, addQty, qInputStr) {

       let newVal = isManualEdit ? addQty : currentVal + addQty;

       let isRafaga = document.getElementById('toggleRafaga') && document.getElementById('toggleRafaga').checked;

           playSound('success');

           saveRowSpecific(item.row, newVal, document.getElementById('obs-'+item.row)?.value || "");
           updateGamification(false);

    function handleSeriesScan(e) {
      if(e.key === 'Enter') {
        this.blur();
        if (!checkAuth()) { document.getElementById('scanInput').value = ""; return; }
        let val = e.target.value.trim().toUpperCase(); if(!val) return; e.target.value = ""; 
        if(SERIES_DATA.some(i => i.serie === val)) { playSound('error'); return Swal.fire("Duplicado", "Serie ya escaneada.", "warning"); }

        const tempId = Date.now(); SERIES_DATA.push({ id: tempId, serie: val, type: 'new', loading: true }); renderSeriesList(); 
        google.script.run.withSuccessHandler(res => {
           const idx = SERIES_DATA.findIndex(i => i.id === tempId);
           if(idx > -1) {
              const item = SERIES_DATA[idx]; item.loading = false;
              if (res.serieReal && res.serieReal !== val) item.serie = res.serieReal; 
              if(res.duplicado) { playSound('error'); item.duplicate = true; item.desc = "DUP: "+(res.datos.pos||"?"); if(res.datos.pos) document.getElementById('displayOrigin').value = res.datos.pos; } 
              else if(res.encontrado) { 
                  item.existe = true; item.sku = res.datos.sku; item.desc = res.datos.desc; item.posOriginal = res.datos.pos; document.getElementById('displayOrigin').value = res.datos.pos; 
                  let tPos = document.getElementById('targetPos').value.trim().toUpperCase(); 
                  let referencePos = tPos || CURRENT_POS;
                  if (referencePos && res.datos.pos !== referencePos) { playSound('error'); item.wrongPos = true; } else { playSound('beep'); showToast('Serie registrada'); } 
              } 
              else { playSound('beep'); item.existe = false; item.sku = "SIN DATOS"; item.desc = "No registrado"; document.getElementById('displayOrigin').value = "---"; }
              renderSeriesList(); 
           }
        }).validarSerieCompleta(val, CURRENT_FILE_ID);
      }
    }

    function processBackgroundQueue() {
       if(PROCESSING_QUEUE.length === 0) { IS_PROCESSING_BG = false; document.getElementById('syncStatus').style.opacity = 0; return; }
       IS_PROCESSING_BG = true; document.getElementById('syncStatus').style.opacity = 1; 
       
       const job = PROCESSING_QUEUE.shift(); 
       
       const payload = job.items.map(i => ({ 
           serie: i.serie, 
           producto: (!i.existe && job.alt) ? job.alt.sku : (i.sku || "SIN_SKU"), 
           posicion: i.posOriginal || CURRENT_POS, // <-- ORIGEN: Para que el backend sepa qué fila editar
           nueva_posicion: job.target || "",       // <-- DESTINO FINAL: La información que va a la columna AC
           esDesconocido: !i.existe 
       }));
       
       google.script.run
       .withSuccessHandler(res => { 
           if(res.exito) { 
               showToast("Series Guardadas"); 
               forceSync(); 
           } 
           processBackgroundQueue(); 
       })
       .withFailureHandler(err => { 
           processBackgroundQueue(); 
       })
       .procesarGuardadoInventario(payload, job.alt, USER_EMAIL, CURRENT_FILE_ID);
    }

    function filtrarTablaAnalisis(filtro) {
        CURRENT_ANALYSIS_FILTER = filtro;
        const badge = document.getElementById('filtroActivoBadge');
        
        if(filtro === 'TODOS') { badge.style.display = 'none'; } 
        else { badge.style.display = 'inline-block'; badge.innerText = `Filtro: ${filtro} ✖`; }

        let tbody = document.getElementById('bodyAnalysis'); tbody.innerHTML = "";
        let data = window.GLOBAL_KPI_DATA.discrepancias;
        
        // Aplicamos el filtro si no es 'TODOS'
        if(filtro === 'FALTANTES') data = data.filter(d => d.estado === 'FALTANTE');
        else if(filtro === 'SOBRANTES') data = data.filter(d => d.estado === 'SOBRANTE');
        else if(filtro !== 'TODOS') data = data.filter(d => d.sku === filtro); // Filtro por SKU

        if(data.length === 0 && filtro === 'TODOS') {
            tbody.innerHTML = "<tr><td colspan='8' style='text-align:center; padding:40px; color:var(--success); font-weight:800; font-size: 14px;'><i class='bi bi-shield-check d-block fs-1 mb-2'></i>Inventario Perfecto. Sin Desfases.</td></tr>";

        } else { 
            data.forEach(d => { 
                let cl = d.estado === 'FALTANTE' ? 'var(--danger)' : 'var(--accent)';

                let iconoFoto = "";

                    <td><strong style="color:var(--primary);">${d.pos}</strong></td>
                    <td class="data-mono">${d.sku}</td>
                    <td style="font-weight:500; max-width:150px; overflow:hidden; text-overflow:ellipsis;" title="${d.desc}">${d.desc}</td>
                    <td><span class="badge" style="background:${cl}; color:white; border:none;">${d.estado}</span></td>
                    <td style="text-align:center; font-weight:bold; color:var(--secondary);">${d.qTeo}</td>
                    <td style="text-align:center; font-weight:bold; color:var(--primary);">${d.qFis}</td>

    // FUNCIÓN PARA EXPORTAR EL REPORTE DE DISCREPANCIAS A PDF PROFESIONAL
   function exportarReportePDF() {
    if (!window.GLOBAL_KPI_DATA) {
        showToast("No hay datos cargados en el Dashboard para exportar.");
        return;
    }

    showToast("Generando reporte ejecutivo calibrado...");

    const { jsPDF } = window.jspdf;
    const doc = new jsPDF('landscape', 'mm', 'a4');

    let kpis = window.GLOBAL_KPI_DATA.kpis;
    let filtroActual = CURRENT_ANALYSIS_FILTER;
    let todasLasDiscrepancias = window.GLOBAL_KPI_DATA.discrepancias || [];

    // Trazabilidad de ajustes: Excluimos del reporte de activas lo resuelto
    const discrepanciasActivas = todasLasDiscrepancias.filter(d => 
        d.estado !== 'CORR. AJUSTADO' && d.estado !== 'CORRECTO' && d.estado !== 'AJUSTADO'
    );

    let dataFiltrada = [...todasLasDiscrepancias];
    if (filtroActual === 'FALTANTES') dataFiltrada = dataFiltrada.filter(d => d.estado === 'FALTANTE');
    else if (filtroActual === 'SOBRANTES') dataFiltrada = dataFiltrada.filter(d => d.estado === 'SOBRANTE');
    else if (filtroActual !== 'TODOS') dataFiltrada = dataFiltrada.filter(d => d.sku === filtroActual);

    // --- ENCABEZADO GERENCIAL COMPACTO ---
    doc.setFont("helvetica", "bold");
    doc.setFontSize(14);
    doc.setTextColor(15, 23, 42);
    doc.text("REPORTE DE AUDITORÍA DE INVENTARIO Y TRAZABILIDAD", 14, 15);

    doc.setFontSize(8);
    doc.setFont("helvetica", "normal");
    doc.setTextColor(100, 116, 139);
    doc.text(`Fecha de Emisión: ${new Date().toLocaleString()}  |  Filtro Control: ${filtroActual}`, 14, 20);

    // --- SECCIÓN I: CUADRO DE RESUMEN EJECUTIVO (COMO EXCEL) ---
    doc.setFontSize(10);
    doc.setFont("helvetica", "bold");
    doc.setTextColor(15, 23, 42);
    doc.text("I. RESUMEN DE CONTROL DE AJUSTES", 14, 27);

    const faltantesCount = discrepanciasActivas.filter(d => d.estado === 'FALTANTE').length;
    const sobrantesCount = discrepanciasActivas.filter(d => d.estado === 'SOBRANTE').length;
    const totalDifPendientes = discrepanciasActivas.length; 
    
    const pctFalt = kpis.refTotal > 0 ? ((faltantesCount / kpis.refTotal) * 100).toFixed(2) + '%' : '0.00%';
    const pctSobr = kpis.refTotal > 0 ? ((sobrantesCount / kpis.refTotal) * 100).toFixed(2) + '%' : '0.00%';

    doc.autoTable({
        startY: 31,
        theme: 'grid',
        head: [[
            { content: 'REFERENCIAS', colSpan: 2, styles: { halign: 'center', fillColor: [30, 41, 59], textColor: [255,255,255], fontStyle: 'bold' } },
            { content: 'NETOS', styles: { halign: 'center', fillColor: [30, 41, 59], textColor: [255,255,255], fontStyle: 'bold' } },
            { content: 'DISCREPANCIAS ACTIVAS', colSpan: 2, styles: { halign: 'center', fillColor: [194, 120, 3], textColor: [255,255,255], fontStyle: 'bold' } }
        ]],
        body: [
            [
                { content: 'Referencias Totales a Contar' }, { content: '', styles: { cellWidth: 5 } },
                { content: kpis.refTotal, styles: { halign: 'right', fontStyle: 'bold' } },
                { content: 'Faltantes', styles: { fillColor: [254, 243, 199], halign: 'center', fontStyle: 'bold', textColor: [180, 83, 9] } },
                { content: 'Sobrantes', styles: { fillColor: [254, 243, 199], halign: 'center', fontStyle: 'bold', textColor: [180, 83, 9] } }
            ],
            [
                { content: 'Referencias Validadas' }, { content: '' },
                { content: kpis.refValidadas, styles: { halign: 'right' } },
                { content: faltantesCount, styles: { halign: 'center' } },
                { content: sobrantesCount, styles: { halign: 'center' } }
            ],
            [
                { content: 'Referencias Con Diferencia Pendiente' }, { content: '' },
                { content: totalDifPendientes, styles: { halign: 'right', fontStyle: 'bold', textColor: totalDifPendientes > 0 ? [220, 38, 38] : [22, 163, 74] } },
                { content: pctFalt, styles: { halign: 'center' } },
                { content: pctSobr, styles: { halign: 'center' } }
            ],
            [
                { content: 'Efectividad Operativa Final' }, { content: '' },
                { content: (totalDifPendientes === 0 ? "100.00%" : kpis.efectividadRef + '%'), styles: { halign: 'right', fontStyle: 'bold', textColor: [22, 163, 74] } },
                { content: '0.00%', styles: { halign: 'center', textColor: [150, 150, 150] } },
                { content: '0.00%', styles: { halign: 'center', textColor: [150, 150, 150] } }
            ]
        ],
        styles: { fontSize: 8, cellPadding: 1.8, lineColor: [200, 200, 200], lineWidth: 0.1, textColor: [50, 50, 50] },
        columnStyles: { 0: { cellWidth: 70 }, 1: { cellWidth: 5 }, 2: { cellWidth: 30 }, 3: { cellWidth: 45 }, 4: { cellWidth: 45 } },
        margin: { left: 14 }
    });

    let currentY = doc.lastAutoTable.finalY + 8;

    // --- SECCIÓN II: DETALLE DE DISCREPANCIAS ---
    doc.setFontSize(10);
    doc.setFont("helvetica", "bold");
    doc.setTextColor(15, 23, 42);
    doc.text("II. DETALLE DE SKU Y ESTADOS DE RESOLUCIÓN", 14, currentY);

    let tableRows = [];
    if (dataFiltrada.length === 0) {
        tableRows = [['-', '-', 'SIN REGISTROS PENDIENTES BAJO EL FILTRO SELECCIONADO', '-', '-', '-', '-', '-', '-', '-', '-']];
    } else {
        tableRows = dataFiltrada.map(item => [
            item.pos || "-",
            item.sku || "-",
            (item.desc || "").substring(0, 22),
            item.estado || "-",
            item.qTeo !== undefined ? item.qTeo : "0",
            item.qFis !== undefined ? item.qFis : "0",
            item.dif !== undefined ? item.dif : "0",
            item.conteo1 || "",
            item.resolucion || item.conteo2 || item.conteo3 || "AJUSTADO",
            item.motivo || item.justificacion || "Validación Física",
            item.nuevaPos || "N/A"
        ]);
    }

    doc.autoTable({
        startY: currentY + 3,
        head: [['POS', 'SKU', 'DESCRIPCIÓN', 'ESTADO', 'TEO', 'FIS', 'DIF', 'C.1', 'RESOLUCIÓN', 'OBSERVACIONES', 'NVA POS']],
        body: tableRows,
        theme: 'grid',
        headStyles: { fillColor: [51, 65, 85], textColor: [255, 255, 255], fontStyle: 'bold', fontSize: 7.5 },
        styles: { fontSize: 6.5, cellPadding: 1.2, valign: 'middle' },
        columnStyles: {
            0: { fontStyle: 'bold', cellWidth: 15 },
            1: { fontStyle: 'bold', cellWidth: 22 },
            2: { cellWidth: 35 },
            3: { halign: 'center', fontStyle: 'bold', cellWidth: 25 },
            4: { halign: 'center', cellWidth: 10 },
            5: { halign: 'center', cellWidth: 10 },
            6: { halign: 'center', fontStyle: 'bold', cellWidth: 12 },
            7: { halign: 'center', cellWidth: 15 },
            8: { cellWidth: 25 },
            9: { cellWidth: 40 },
            10: { halign: 'center', cellWidth: 15 }
        },
        didParseCell: function(dataRow) {
            if (dataRow.section === 'body') {
                let cellStatus = dataRow.row.raw[3];
                if (dataRow.column.index === 3) {
                    if (cellStatus === 'FALTANTE') dataRow.cell.styles.textColor = [220, 38, 38]; 
                    else if (cellStatus === 'SOBRANTE') dataRow.cell.styles.textColor = [37, 99, 235]; 
                    else dataRow.cell.styles.textColor = [22, 163, 74]; 
                }
                if (dataRow.column.index === 6) {
                    let difVal = parseFloat(dataRow.row.raw[6]);
                    if (difVal < 0) dataRow.cell.styles.textColor = [220, 38, 38];
                    else if (difVal > 0) dataRow.cell.styles.textColor = [37, 99, 235];
                    else dataRow.cell.styles.textColor = [22, 163, 74];
                }
            }
        }
    });

    // --- SECCIÓN III: MAPA TRIDIMENSIONAL (CAPTURA EN VIVO CON RE-RENDER) ---
    try {
        const mapaCanvas = document.querySelector('canvas');
        if (mapaCanvas) {
            // Si las variables globales están activas, forzamos el renderizado instantáneo:
            if (window.renderer && window.scene && window.camera) {
                window.renderer.render(window.scene, window.camera);
            } else if (window.GLOBAL_RENDERER && window.GLOBAL_SCENE && window.GLOBAL_CAMERA) {
                window.GLOBAL_RENDERER.render(window.GLOBAL_SCENE, window.GLOBAL_CAMERA);
            }

            const mapaImgData = mapaCanvas.toDataURL('image/png');
            
            doc.addPage();
            doc.setFont("helvetica", "bold");
            doc.setFontSize(11);
            doc.setTextColor(15, 23, 42);
            doc.text("III. ANEXO VISUAL: CONFIGURACIÓN EN MAPA 3D DEL ALMACÉN", 14, 15);
            
            doc.setFontSize(8);
            doc.setFont("helvetica", "normal");
            doc.setTextColor(100, 116, 139);
            doc.text("Estado tridimensional de ocupación, slots críticos y pasillos auditados.", 14, 20);

            let deAncho = 269; 
            let deAlto = (mapaCanvas.height * deAncho) / mapaCanvas.width;
            if (deAlto > 155) { 
                deAlto = 155;
                deAncho = (mapaCanvas.width * deAlto) / mapaCanvas.height;
            }

            doc.addImage(mapaImgData, 'PNG', (297 - deAncho) / 2, 26, deAncho, deAlto);
        }
    } catch (err) {
        console.warn("Anexo del mapa omitido en este ciclo: ", err);
    }

    const stringFecha = new Date().toISOString().slice(0, 10);
    doc.save(`Reporte_Auditoria_WMS_Gerencial_${stringFecha}.pdf`);
    showToast("¡Reporte Gerencial descargado con éxito!");
}
