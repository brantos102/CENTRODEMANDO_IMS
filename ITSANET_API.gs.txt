 * end-point g_apidepot. (El cliente Hycite se atenderá más adelante.)
 *
 * Pegar como archivo .gs NUEVO en el proyecto "Centro de Mando" (ya tiene script.external_request).
 * No modifica nada del flujo actual.
 *
 * SEGURIDAD: la CLAVE no va en el código. Se lee de ScriptProperties. Ejecuta UNA vez
 * setupClaveItsanet() con tu clave real (en TU proyecto), o cárgala manualmente en
 * Configuración del proyecto -> Propiedades del script -> ITSANET_PASS.
 */

var ITSANET_USERID = "FINI";
var ITSANET_BASE   = "https://ec.itsanet.com/g_apidepot";

// (1) Ejecuta UNA vez con tu clave real (reemplaza el placeholder). No subas la clave al repo.
function setupClaveItsanet() {
  PropertiesService.getScriptProperties().setProperty('ITSANET_PASS', 'PEGAR_AQUI_LA_CLAVE_FINI');
  Logger.log('Listo. Pon la clave real en ITSANET_PASS antes de ejecutar las pruebas.');
}

/**
 * Centraliza el control y caché del token.
 * REGLA: un solo token por día. Si ya existe el de hoy, se reutiliza; si no, se genera uno nuevo.
 */
function obtenerORenovarTokenWMS() {
  var sp = PropertiesService.getScriptProperties();
  var hoy = Utilities.formatDate(new Date(), "GMT-5", "yyyy-MM-dd");
  var tokenGuardado = sp.getProperty('ITSANET_TOKEN');
  var fechaGuardada = sp.getProperty('FECHA_TOKEN');

  if (tokenGuardado && fechaGuardada === hoy) {
    return tokenGuardado; // ya hay token válido de hoy -> no se crea otro
  }

  var pass = sp.getProperty('ITSANET_PASS');
  if (!pass || pass.indexOf('PEGAR_AQUI') === 0) {
    throw new Error('Falta ITSANET_PASS. Ejecuta setupClaveItsanet() con la clave real o cárgala en Propiedades del script.');
  }

  var resToken = UrlFetchApp.fetch(ITSANET_BASE + "/gettoken", {
    "method": "get",
    "headers": { "userid": ITSANET_USERID, "password": pass },
    "muteHttpExceptions": true
  });
  var code = resToken.getResponseCode();
  var body = resToken.getContentText();
  var json = {}; try { json = JSON.parse(body); } catch (e) {}
  var nuevoToken = json.token || json.Token || json.Result || json.result;

  if (nuevoToken) {
    sp.setProperty('ITSANET_TOKEN', String(nuevoToken).trim());
    sp.setProperty('FECHA_TOKEN', hoy);
    return String(nuevoToken).trim();
  } else {
    throw new Error("No se pudo renovar el token diario con Itsanet. HTTP " + code + " -> " + body.substring(0, 300));
  }
}

// (2) Consulta el STOCK con el token del día. Devuelve {code, body}.
function consultarStockItsanet() {
  var token = obtenerORenovarTokenWMS();
  var res = UrlFetchApp.fetch(ITSANET_BASE + "/getstock", {
    "method": "get",
    "headers": { "token": String(token).trim() },
    "muteHttpExceptions": true
  });
  return { code: res.getResponseCode(), body: res.getContentText() };
}

// (3) === EJECUTA ESTA para probar. Revisa: Ver -> Registro de ejecución ===
function probarConexionItsanet() {
  try {
    var tok = obtenerORenovarTokenWMS();
    Logger.log("OK TOKEN del día: " + String(tok).substring(0, 16) + "...");
    var r = consultarStockItsanet();
    Logger.log("getstock -> HTTP " + r.code);
    var arr = null;
    try {
      var j = JSON.parse(r.body);
      arr = Array.isArray(j) ? j
          : (j && Array.isArray(j.Result)) ? j.Result
          : (j && Array.isArray(j.result)) ? j.result
          : (j && Array.isArray(j.data)) ? j.data : null;
    } catch (e) {}
    if (arr) {
      Logger.log("OK Filas de stock: " + arr.length);
      Logger.log("Ejemplo[0]: " + JSON.stringify(arr[0]));
    } else {
      Logger.log("Respuesta (primeros 1500): " + String(r.body).substring(0, 1500));
    }
    return r;
  } catch (e) {
    Logger.log("ERROR de conexión ITSANET: " + e.message);
    return { error: e.message };
  }
}
