 * Objetivo de esta fase: conectar y OBTENER DATOS (stock) de forma efectiva. Credencial FINI,
 * end-point g_apidepot. (El cliente Hycite se atenderá más adelante.)
 *
 * Pegar como archivo .gs NUEVO en el proyecto "Centro de Mando" (ya tiene script.external_request).
 * No modifica nada del flujo actual.
 *
 * SEGURIDAD: la CLAVE no va en el código. Se lee de ScriptProperties. Ejecuta UNA vez
 * setupClaveItsanet() con tu clave real (en TU proyecto), o cárgala manualmente en
 * Configuración del proyecto -> Propiedades del script -> ITSANET_PASS.
 */

var ITSANET_USERID = "FINI";
var ITSANET_BASE   = "https://ec.itsanet.com/g_apidepot";

// (1) Ejecuta UNA vez con tu clave real (reemplaza el placeholder). No subas la clave al repo.
function setupClaveItsanet() {
  PropertiesService.getScriptProperties().setProperty('ITSANET_PASS', 'PEGAR_AQUI_LA_CLAVE_FINI');
  Logger.log('Listo. Pon la clave real en ITSANET_PASS antes de ejecutar las pruebas.');
}

/**
 * Centraliza el control y caché del token.
 * REGLA: un solo token por día. Si ya existe el de hoy, se reutiliza; si no, se genera uno nuevo.
 */
function obtenerORenovarTokenWMS() {
  var sp = PropertiesService.getScriptProperties();
  var hoy = Utilities.formatDate(new Date(), "GMT-5", "yyyy-MM-dd");
  var tokenGuardado = sp.getProperty('ITSANET_TOKEN');
  var fechaGuardada = sp.getProperty('FECHA_TOKEN');

  if (tokenGuardado && fechaGuardada === hoy) {
    return tokenGuardado; // ya hay token válido de hoy -> no se crea otro
  }

  var pass = sp.getProperty('ITSANET_PASS');
  if (!pass || pass.indexOf('PEGAR_AQUI') === 0) {
    throw new Error('Falta ITSANET_PASS. Ejecuta setupClaveItsanet() con la clave real o cárgala en Propiedades del script.');
  }

  var resToken = UrlFetchApp.fetch(ITSANET_BASE + "/gettoken", {
    "method": "get",
    "headers": { "userid": ITSANET_USERID, "password": pass },
    "muteHttpExceptions": true
  });
  var code = resToken.getResponseCode();
  var body = resToken.getContentText();
  var json = {}; try { json = JSON.parse(body); } catch (e) {}
  var nuevoToken = json.token || json.Token || json.Result || json.result;

  if (nuevoToken) {
    sp.setProperty('ITSANET_TOKEN', String(nuevoToken).trim());
    sp.setProperty('FECHA_TOKEN', hoy);
    return String(nuevoToken).trim();
  } else {
    throw new Error("No se pudo renovar el token diario con Itsanet. HTTP " + code + " -> " + body.substring(0, 300));
  }
}

// (2) Consulta el STOCK con el token del día. Devuelve {code, body}.
function consultarStockItsanet() {
  var token = obtenerORenovarTokenWMS();
  var res = UrlFetchApp.fetch(ITSANET_BASE + "/getstock", {
    "method": "get",
    "headers": { "token": String(token).trim() },
    "muteHttpExceptions": true
  });
  return { code: res.getResponseCode(), body: res.getContentText() };
}

// (3) === EJECUTA ESTA para probar. Revisa: Ver -> Registro de ejecución ===
function probarConexionItsanet() {
  try {
    var tok = obtenerORenovarTokenWMS();
    Logger.log("OK TOKEN del día: " + String(tok).substring(0, 16) + "...");
    var r = consultarStockItsanet();
    Logger.log("getstock -> HTTP " + r.code);
    var arr = null;
    try {
      var j = JSON.parse(r.body);
      arr = Array.isArray(j) ? j
          : (j && Array.isArray(j.Result)) ? j.Result
          : (j && Array.isArray(j.result)) ? j.result
          : (j && Array.isArray(j.data)) ? j.data : null;
    } catch (e) {}
    if (arr) {
      Logger.log("OK Filas de stock: " + arr.length);
      Logger.log("Ejemplo[0]: " + JSON.stringify(arr[0]));
    } else {
      Logger.log("Respuesta (primeros 1500): " + String(r.body).substring(0, 1500));
    }
    return r;
  } catch (e) {
    Logger.log("ERROR de conexión ITSANET: " + e.message);
    return { error: e.message };
  }
}

/**
 * (4) EXTRACCIÓN para crear inventario: trae el stock, lo FILTRA por cliente y (opcional) por una
 * lista de códigos/referencias, y lo MAPEA a las columnas de la PLANILLA DE CONTEO FÍSICO.
 *   - codClienteFiltro: p.ej. "FINI" (vacío = todos).
 *   - listaCodigos: arreglo o texto (separado por coma/salto) de COD. PRODUCTO a incluir (vacío = todos).
 * Devuelve { ok, total, filas: [...] } donde cada fila trae las claves que usa el importador.
 * NOTA: la consolidación final (sumar cantidades por SKU+posición, reglas de series) se hace en
 * el paso de importación del Centro de Mando, igual que con el CSV (reutilizando esa lógica).
 */
function extraerStockItsanet(codClienteFiltro, listaCodigos) {
  var r = consultarStockItsanet();
  if (r.code !== 200) return { ok: false, code: r.code, error: 'getstock HTTP ' + r.code, filas: [] };
  var j; try { j = JSON.parse(r.body); } catch (e) { return { ok:false, error:'JSON inválido', filas:[] }; }
  var arr = Array.isArray(j) ? j : (j.Result || j.result || j.data || []);

  var cliF = String(codClienteFiltro || '').trim().toUpperCase();
  var setCods = null;
  if (listaCodigos) {
    var raw = Array.isArray(listaCodigos) ? listaCodigos.join(',') : String(listaCodigos);
    var lista = raw.split(/[\s,;\n\r\t]+/).map(function(s){ return String(s).trim().toUpperCase(); }).filter(function(s){ return s.length > 0; });
    if (lista.length) { setCods = {}; lista.forEach(function(c){ setCods[c] = true; }); }
  }

  function val(o, k) { return (o[k] === null || o[k] === undefined) ? '' : String(o[k]).trim(); }

  var filas = [];
  for (var i = 0; i < arr.length; i++) {
    var o = arr[i] || {};
    var cod = val(o, 'COD. PRODUCTO');
    if (!cod) continue;
    if (cliF && val(o, 'COD. CLIENTE').toUpperCase() !== cliF) continue;
    if (setCods && !setCods[cod.toUpperCase()]) continue;

    // Posición: NAVE-CALLE-COLUMNA-NIVEL (ajustar el formato si difiere del usado por el WMS)
    var pos = [val(o,'NAVE'), val(o,'CALLE'), val(o,'COLUMNA'), val(o,'NIVEL')].filter(function(x){ return x !== ''; }).join('-');

    filas.push({
      sku:        cod,
      desc:       val(o, 'DESCRIPCION'),
      unidad:     val(o, 'UNIDAD'),
      cantidad:   parseFloat(val(o, 'CANTIDAD')) || 0,
      posicion:   pos,
      serie:      val(o, 'NRO. SERIE'),
      lote:       val(o, 'NRO. LOTE'),
      nDespacho:  val(o, 'NRO. DESPACHO'),
      nPartida:   val(o, 'NRO. PARTIDA'),
      catLog:     val(o, 'CAT. LOG.'),
      estMerc:    val(o, 'EST. MERC.'),
      cliente:    val(o, 'COD. CLIENTE'),
      familia:    val(o, 'FAMILIA')
    });
  }
  return { ok: true, total: filas.length, filas: filas };
}

// (5) Prueba de extracción (revisa el Registro de ejecución). Ajusta cliente/códigos según necesites.
function probarExtraccionItsanet() {
  var res = extraerStockItsanet('FINI', '');   // p.ej. extraerStockItsanet('FINI', '9990, 1234')
  Logger.log('Filas extraídas: ' + res.total);
  if (res.filas && res.filas.length) Logger.log('Fila[0] mapeada: ' + JSON.stringify(res.filas[0]));
  return res;
}

/* ===========================================================================
   PUENTE CON LA CREACIÓN DE INVENTARIO (reusa validarCSVAvanzado, sin duplicar)
   El stock se convierte a filas de 17 columnas IDÉNTICAS a las del CSV y se pasa
   por la MISMA validación/consolidación del wizard.
   =========================================================================== */

// Índices del array de 17 columnas que consume validarCSVAvanzado.
// CONFIRMADOS por la validación: 2,3,4,5,6,7,8,16.  Los demás (descripción, unidad, etc.)
// quedan marcados como "CONFIRMAR" hasta tener el encabezado real del CSV del wizard.
function _filaCsvDesdeStock(o) {
  function v(k){ return (o[k]===null||o[k]===undefined) ? "" : String(o[k]).trim(); }
  var fila = [];
  for (var n=0;n<17;n++) fila.push("");
  // --- Índices USADOS por validarCSVAvanzado (correctos) ---
  fila[2]  = v('NRO. DESPACHO');                                   // N. Despacho
  fila[3]  = v('NRO. PARTIDA');                                    // N. Partida
  fila[4]  = v('COD. CLIENTE');                                    // Cliente
  fila[5]  = [v('NAVE'),v('CALLE'),v('COLUMNA'),v('NIVEL')].filter(function(x){return x!=="";}).join('-'); // Posición
  fila[6]  = v('COD. PRODUCTO');                                   // SKU
  fila[7]  = v('NRO. LOTE');                                       // Lote
  fila[8]  = v('NRO. SERIE');                                      // Serie
  fila[16] = parseFloat(v('CANTIDAD')) || 0;                       // Cantidad
  // --- Campos de paso (CONFIRMAR su índice con el encabezado del CSV del wizard) ---
  fila[0]  = v('DESCRIPCION');   // ¿col 0?  descripción
  fila[1]  = v('UNIDAD');        // ¿col 1?  unidad
  fila[9]  = v('CAT. LOG.');     // ¿col 9?  categoría logística
  fila[10] = v('EST. MERC.');    // ¿col 10? estado mercadería
  return fila;
}

function _setCodigos(listaCodigos) {
  if (!listaCodigos) return null;
  var raw = Array.isArray(listaCodigos) ? listaCodigos.join(',') : String(listaCodigos);
  var lista = raw.split(/[\s,;\n\r\t]+/).map(function(s){ return String(s).trim().toUpperCase(); }).filter(function(s){ return s.length>0; });
  if (!lista.length) return null;
  var set = {}; lista.forEach(function(c){ set[c]=true; }); return set;
}

// Trae el stock y lo devuelve como filas de 17 columnas (formato CSV del wizard).
// Filtra por lista de códigos si se indica (variantes con sufijo incluidas si incluirVariantes=true).
function extraerStockComoFilasCSV(listaCodigos, incluirVariantes) {
  var r = consultarStockItsanet();
  if (r.code !== 200) throw new Error('getstock HTTP ' + r.code + ' -> ' + String(r.body).substring(0,200));
  var j = JSON.parse(r.body);
  var arr = Array.isArray(j) ? j : (j.Result || j.result || j.data || []);
  var setCods = _setCodigos(listaCodigos);
  var filas = [];
  for (var i=0;i<arr.length;i++){
    var o = arr[i] || {};
    var cod = (o['COD. PRODUCTO']==null?'':String(o['COD. PRODUCTO']).trim()).toUpperCase();
    if (!cod) continue;
    if (setCods) {
      var match = !!setCods[cod];
      if (!match && incluirVariantes) { // variante con sufijo: el código empieza por uno de la lista (ej. 9990_U)
        for (var k in setCods) { if (cod.indexOf(k) === 0) { match = true; break; } }
      }
      if (!match) continue;
    }
    filas.push(_filaCsvDesdeStock(o));
  }
  return filas;
}

// ORQUESTADOR para el wizard: extrae de la API y pasa por la MISMA validación del CSV.
// Devuelve { datosLimpios, reporte } igual que la carga de CSV (mismas reglas/consolidación).
function previsualizarStockItsanet(clienteSeleccionado, listaCodigos, incluirVariantes) {
  var filas = extraerStockComoFilasCSV(listaCodigos, incluirVariantes);
  if (!filas.length) throw new Error('La API no devolvió filas para los códigos indicados.');
  return validarCSVAvanzado(filas, clienteSeleccionado); // reusa toda la lógica del wizard
}
